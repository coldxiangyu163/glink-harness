#!/usr/bin/env python3
"""GLINK-ACP-10 真实冒烟：blueprint run 端到端（inproc、pi 面）。

在进程内起真实宿主（临时数据根、随机 loopback 端口），从产品 HTTP API 用
内置 `deep-research` 蓝图创建一个 run，轮询终态并回收 report.md。模型凭据从
操作员 `.env` 读取（与本次任务相关、不打印、不提交）；执行面为 inproc（不起
Docker/E2B）；退出时只关闭本进程持有的宿主并删除临时数据根。

用法：PYTHONPATH=<worktree>/host python3 /tmp/blueprint_smoke.py
退出码：0 = run 终态为 finished、记录携带蓝图与阶段元数据、report.md 可下载。
"""

from __future__ import annotations

import json
import shutil
import sys
import tempfile
import threading
import time
import urllib.error
import urllib.request
from pathlib import Path

REPO_ROOT = Path("~/.herdr/worktrees/pi-glink/blueprint-inc1-20260925")
OPERATOR_ENV = Path("~/AI/git/pi-glink/.env")
sys.path.insert(0, str(REPO_ROOT / "host"))

from pi_glink.app import create_server  # noqa: E402
from pi_glink.config import load_config  # noqa: E402

PROMPT = "请用一句话回答：1+1 等于几？并按系统提示里的工作约定交付 report.md。"
TIMEOUT_SECONDS = 600.0


def main() -> int:
    scratch = Path(tempfile.mkdtemp(prefix="pi-glink-blueprint-smoke."))
    overrides = {
        "LOCAL_WEB_HOST": "127.0.0.1",
        "LOCAL_WEB_PORT": "0",
        "LOCAL_WEB_EXECUTOR": "inproc",
        "LOCAL_WEB_DATA_ROOT": str(scratch / "data"),
        "LOCAL_WEB_TASK_TIMEOUT_SECONDS": str(int(TIMEOUT_SECONDS)),
        "PI_GLINK_TENANCY": "local",
        "PI_GLINK_AUTH_MODE": "gateway",
        "PI_GLINK_MODEL_CREDENTIALS": "tenant",
        "PI_GLINK_HARNESS": "pi",
        "PI_CODING_AGENT_DIR": str(REPO_ROOT / "agent"),
        "PI_GLINK_BLUEPRINTS_DIR": str(REPO_ROOT / "blueprints"),
        "PI_GLINK_WEB_SEARCH_CONFIG": str(scratch / "absent-web-search.json"),
    }
    config = load_config(env=overrides, repo_root=REPO_ROOT, env_file=OPERATOR_ENV)
    server = create_server(config)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    port = server.server_address[1]
    base = f"http://127.0.0.1:{port}"
    started = time.monotonic()

    def api(path: str, *, method: str = "GET", body: dict | None = None):
        payload = json.dumps(body).encode("utf-8") if body is not None else None
        request = urllib.request.Request(f"{base}{path}", data=payload, method=method)
        if payload is not None:
            request.add_header("Content-Type", "application/json")
        try:
            with urllib.request.urlopen(request, timeout=60) as response:
                return response.status, response.read()
        except urllib.error.HTTPError as error:
            return error.code, error.read()

    try:
        print(f"[host] blueprint dir={config.blueprints_dir} executor={config.executor}")
        status, raw = api(
            "/api/runs", method="POST", body={"prompt": PROMPT, "blueprint": "deep-research"}
        )
        if status != 201:
            print(f"[fail] POST /api/runs status={status} body={raw[:400]!r}")
            return 1
        run = json.loads(raw)["run"]
        run_id = run["runId"]
        print(
            f"[create] status=201 runId={run_id} blueprint={run['blueprint']} "
            f"stages={[stage['id'] for stage in run['blueprintStages']]}"
        )

        deadline = time.monotonic() + TIMEOUT_SECONDS
        final = run
        while time.monotonic() < deadline:
            _status, raw = api(f"/api/runs/{run_id}")
            final = json.loads(raw)["run"]
            if final["status"] != "running":
                break
            time.sleep(2.0)

        elapsed = round(time.monotonic() - started, 1)
        print(
            "[terminal] "
            + json.dumps(
                {
                    key: final.get(key)
                    for key in (
                        "status",
                        "stopReason",
                        "blueprint",
                        "blueprintStages",
                        "eventCount",
                        "hasReport",
                        "durationSeconds",
                        "modelStopReason",
                        "error",
                    )
                },
                ensure_ascii=False,
            )
        )
        print(f"[elapsed] {elapsed}s")

        events_path = Path(final.get("_unused", "") or (scratch / "data" / "runs" / run_id / "events.jsonl"))
        if events_path.is_file():
            frames = [
                json.loads(line)
                for line in events_path.read_text(encoding="utf-8").splitlines()
                if line.strip()
            ]
            types = {}
            for frame in frames:
                types[frame.get("type")] = types.get(frame.get("type"), 0) + 1
            print(f"[events] count={len(frames)} types={json.dumps(types, ensure_ascii=False)}")
            terminal = frames[-1].get("payload", {}) if frames else {}
            print(
                "[terminal-payload] "
                + json.dumps(
                    {
                        key: terminal.get(key)
                        for key in ("status", "reportStatus", "modelStopReason", "cleanup")
                    },
                    ensure_ascii=False,
                )
            )

        report_text = ""
        if final.get("hasReport"):
            status, raw = api(f"/api/runs/{run_id}/report")
            report_text = raw.decode("utf-8", errors="replace")
            print(f"[report] status={status} bytes={len(raw)}")
            print("[report-excerpt]")
            print(report_text[:1500])
        else:
            print("[report] missing")

        ok = (
            final.get("status") == "finished"
            and final.get("blueprint") == "deep-research"
            and len(final.get("blueprintStages") or []) == 4
            and bool(final.get("hasReport"))
            and "来源与核查" in report_text
        )
        print(f"[verdict] {'PASS' if ok else 'FAIL'}")
        return 0 if ok else 1
    finally:
        server.shutdown()
        server.server_close()
        thread.join(timeout=10)
        shutil.rmtree(scratch, ignore_errors=True)
        print(f"[cleanup] host stopped; temporary data root removed: {scratch}")


if __name__ == "__main__":
    raise SystemExit(main())
